Structural reference only. Generated chapters are independent scenes.

```tsx
import {Circle, makeScene2D} from '@motion-canvas/2d';
import {createSignal, linear} from '@motion-canvas/core';

const CHAPTER_DURATION = 4;
export default makeScene2D(function* (view) {
  const progress = createSignal(0);
  const localTime = () => progress() * CHAPTER_DURATION;
  const state = () => stateAt(localTime());
  view.add(<Circle position={() => [state().position.x, -state().position.y]} />);
  yield* progress(1, CHAPTER_DURATION, linear);
});
```
